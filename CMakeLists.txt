cmake_minimum_required(VERSION 3.25)
project(Gate VERSION 0.1.1 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Gate requires Windows x64 and the MSVC toolchain.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Select the x64 toolchain.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT "$ENV{WindowsSDKVersion}" MATCHES "^10[.]0[.]26100[.]0")
  message(FATAL_ERROR "Use the pinned Windows SDK 10.0.26100.0 (scripts/build.cmd selects it).")
endif()
configure_file(cmake/build-metadata.json.in build-metadata.json @ONLY)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
include(cmake/Dependencies.cmake)
add_library(gate_processing STATIC src/audio/Processing.cpp src/audio/Resampler.cpp src/audio/Voice.cpp)
target_include_directories(gate_processing PUBLIC src)
target_include_directories(gate_processing SYSTEM PUBLIC third_party/q/q_lib/include third_party/q/infra/include third_party/signalsmith-stretch third_party/signalsmith-linear/include)
target_link_libraries(gate_processing PUBLIC rnnoise speex_resampler)
add_library(gate_platform STATIC src/audio/Devices.cpp src/audio/Engine.cpp src/audio/ClipPlayer.cpp src/app/Preferences.cpp)
target_include_directories(gate_platform PUBLIC src)
target_link_libraries(gate_platform PUBLIC gate_processing ole32 uuid avrt propsys cfgmgr32 shell32 mfplat mfreadwrite mfuuid)
set(STARTER_SOUND_FILES air-horn.wav drum-roll.wav rimshot.wav buzzer.wav chime.wav sad-trombone.wav)
list(TRANSFORM STARTER_SOUND_FILES PREPEND "${CMAKE_CURRENT_SOURCE_DIR}/resources/sounds/")
set_property(SOURCE resources/StarterSounds.rc PROPERTY OBJECT_DEPENDS "${STARTER_SOUND_FILES}")
add_executable(Gate WIN32 src/app/Main.cpp src/ui/Window.cpp src/ui/Layout.cpp src/ui/Controls.cpp src/ui/VoicePages.cpp src/ui/EmojiPicker.cpp src/ui/Theme.cpp resources/Gate.rc resources/StarterSounds.rc)
target_include_directories(Gate PRIVATE src)
target_link_libraries(Gate PRIVATE gate_platform d2d1 dwrite dwmapi comctl32 shell32 shlwapi windowsapp uxtheme oleacc comdlg32)
target_link_options(Gate PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/Gate.manifest /Brepro)
foreach(target gate_processing gate_platform Gate)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8 /Zc:__cplusplus /Brepro)
endforeach()
include(CTest)
if(BUILD_TESTING)
  target_compile_definitions(gate_platform PRIVATE GATE_DEVELOPER_PROBES)
  add_executable(gate_tests tests/CoreTests.cpp)
  target_link_libraries(gate_tests PRIVATE gate_platform)
  target_compile_options(gate_tests PRIVATE /W4 /EHsc /utf-8 /Zc:__cplusplus)
  add_test(NAME core COMMAND gate_tests)
  add_executable(gate_feature_tests tests/FeatureTests.cpp resources/StarterSounds.rc)
  target_link_libraries(gate_feature_tests PRIVATE gate_platform shell32)
  target_compile_options(gate_feature_tests PRIVATE /W4 /EHsc /utf-8 /Zc:__cplusplus)
  add_test(NAME features COMMAND gate_feature_tests)
  add_executable(gate_probe tests/AudioProbe.cpp)
  target_link_libraries(gate_probe PRIVATE gate_platform)
endif()
install(TARGETS Gate RUNTIME DESTINATION .)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES packaging/README.txt DESTINATION .)
install(FILES third_party/rnnoise/COPYING DESTINATION licenses RENAME RNNoise.txt)
install(FILES third_party/speexdsp/COPYING DESTINATION licenses RENAME SpeexDSP.txt)
install(FILES third_party/q/LICENSE DESTINATION licenses RENAME Cycfi-Q.txt)
install(FILES packaging/licenses/Cycfi-infra.txt DESTINATION licenses)
set(CPACK_PACKAGE_NAME Gate)
set(CPACK_PACKAGE_VENDOR "Gate contributors")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "Gate-${PROJECT_VERSION}-windows-x64")
include(CPack)

install(FILES third_party/signalsmith-stretch/LICENSE.txt DESTINATION licenses RENAME Signalsmith-Stretch.txt)
install(FILES third_party/signalsmith-linear/LICENSE.txt DESTINATION licenses RENAME Signalsmith-Linear.txt)
